#include <iostream>
#include <algorithm>
using namespace std;

int partition(int A[], int low, int high)
{
    int i = low;
    int j = high;
    int pivot = A[low];

    while (i < j)
    {
        while (A[i] <= pivot && i < high)
            i++;

        while (A[j] > pivot)
            j--;

        if (i < j)
            swap(A[i], A[j]);
    }

    swap(A[low], A[j]);
    return j;
}

void QuickSort(int A[], int low, int high)
{
    if (low < high)
    {
        int j = partition(A, low, high);
        QuickSort(A, low, j - 1);
        QuickSort(A, j + 1, high);
    }
}

int main()
{
    int A[50], n;

    cout << "Enter the number of elements: ";
    cin >> n;

    if (n < 1 || n > 50)
    {
        cout << "Please enter a size between 1 and 50.";
        return 0;
    }

    cout << "Enter the elements: ";
    for (int i = 0; i < n; i++)
        cin >> A[i];

    QuickSort(A, 0, n - 1);

    cout << "Sorted array: ";
    for (int i = 0; i < n; i++)
        cout << A[i] << " ";

    return 0;
}
