#include <iostream>
using namespace std;

int main()
{
    int hashTable[10], key, index, choice;

    for (int i = 0; i < 10; i++)
        hashTable[i] = -1;

    while (true)
    {
        cout << "\\n1. Insert";
        cout << "\\n2. Search";
        cout << "\\n3. Display";
        cout << "\\n4. Exit";
        cout << "\\nEnter choice: ";
        cin >> choice;

        if (choice == 1)
        {
            cout << "Enter key: ";
            cin >> key;

            index = key % 10;
            int start = index;

            while (hashTable[index] != -1)
            {
                if (hashTable[index] == key)
                    break;

                index = (index + 1) % 10;

                if (index == start)
                    break;
            }

            if (hashTable[index] == key)
                cout << "Key already exists";
            else if (hashTable[index] == -1)
            {
                hashTable[index] = key;
                cout << "Key inserted at index " << index;
            }
            else
                cout << "Hash table is full";
        }
        else if (choice == 2)
        {
            cout << "Enter key to search: ";
            cin >> key;

            index = key % 10;
            int start = index;

            while (hashTable[index] != -1)
            {
                if (hashTable[index] == key)
                    break;

                index = (index + 1) % 10;

                if (index == start)
                    break;
            }

            if (hashTable[index] == key)
                cout << "Key found at index " << index;
            else
                cout << "Key not found";
        }
        else if (choice == 3)
        {
            cout << "\\nHash Table:\\n";

            for (int i = 0; i < 10; i++)
            {
                cout << i << ": ";

                if (hashTable[i] == -1)
                    cout << "Empty";
                else
                    cout << hashTable[i];

                cout << endl;
            }
        }
        else if (choice == 4)
            break;
        else
            cout << "Invalid choice";
    }

    return 0;
}
